599: def _tmx_import_draft(
600:     record: ResourceRecord,
601:     file_name: str,
602: ) -> TMRecordDraft:
603:     """One exact import draft with lossless TMX prop provenance.
604: 
605:     Parser owns XML safety and the ordered prop grammar.  This adapter only
606:     recognizes LocalCAT-owned semantic props plus the verified MateCAT status
607:     prop; every prop is also retained in canonical JSON form so unknown and
608:     duplicate properties survive a later TMX export.
609:     """
610: 
611:     speaker: str | None = None
612:     context_prev: str | None = None
613:     context_next: str | None = None
614:     file_source: str | None = file_name
615:     provenance: list[tuple[str, str]] = [
616:         ("source", "tmx-import"),
617:         ("file", file_name),
618:     ]
619:     for scope, prop_type, language, value in _tmx_record_props(record):
620:         raw_prop = json.dumps(
621:             (scope, prop_type, language, value),
622:             ensure_ascii=False,
623:             allow_nan=False,
624:             separators=(",", ":"),
625:         )
626:         provenance.append(("tmx.prop", raw_prop))
627:         normalized_type = prop_type.casefold()
628:         if scope == "tu" and normalized_type == "x-localcat-context-prev":
629:             context_prev = value or None
630:         elif scope == "tu" and normalized_type == "x-localcat-context-next":
631:             context_next = value or None
632:         elif scope == "tu" and normalized_type == "x-localcat-speaker":
633:             speaker = value or None
634:         elif scope == "tu" and normalized_type == "x-localcat-file-source":
635:             file_source = value or None
636:         elif normalized_type in {"x-localcat-status", "x-matecat-status"}:
637:             provenance.append(("tmx.status", value))
638:         elif scope == "tu" and normalized_type == "x-localcat-provenance":
639:             try:
640:                 decoded = json.loads(value)
641:             except (TypeError, ValueError, json.JSONDecodeError):
642:                 decoded = None
643:             if (
644:                 type(decoded) is list
645:                 and len(decoded) == 2
646:                 and all(type(item) is str and item for item in decoded)
647:             ):
648:                 provenance.append((decoded[0], decoded[1]))
649: 
650:     return TMRecordDraft(
651:         source_raw=record.source,
652:         target_raw=record.target,
653:         speaker_raw=speaker,
654:         context_prev_raw=context_prev,
655:         context_next_raw=context_next,
656:         file_source=file_source,
657:         provenance=tuple(provenance),
658:     )
